class BuildProject{
  constructor(k,type,pos,rotation=0,variant=null){
    this.kingdom=k;this.type=type;this.pos=pos.clone();this.rotation=rotation;this.done=false;this.id=Math.random().toString(36).slice(2);const bp=instantiateBlueprint(type,variant);this.definition=bp.def;this.variant=bp.vi;this.blocks=bp.blocks;this.meshes=[];
    this.marker=cube('site',v3(pos.x,yAt(pos.x,pos.z)+.04,pos.z),v3(2.0,.08,2.0),mat(`site-${k.id}`,k.bright,.38,k.color.scale(.12)),false);
    this.label=makeWorldLabel(`${bp.def.label}\nFOUNDATION`,this.marker,k.id);
  }
  blockCost(b){return b.r==='iron'?.18:b.r==='stone'?.12:.10}
  worldFor(b){const q=rotateLocal(b.x*CONFIG.buildBlock,b.z*CONFIG.buildBlock,this.rotation);return v3(this.pos.x+q.x,yAt(this.pos.x,this.pos.z)+CONFIG.buildBlock*.5+b.y*CONFIG.buildBlock,this.pos.z+q.z)}
  claim(npc){const b=this.blocks.find(x=>!x.placed&&!x.claimedBy);if(!b)return null;b.claimedBy=npc;const claim={project:this,block:b,world:this.worldFor(b)};return claim;}
  placeClaim(claim){
    const b=claim.block;if(this.done||b.placed)return;const mesh=cube(`build-${this.type}`,claim.world,v3(CONFIG.buildBlock,CONFIG.buildBlock,CONFIG.buildBlock),mats[b.m]||mats.plank,true);mesh.rotation.y=this.rotation;mesh.metadata={kind:'buildingProject',ref:this};shadow.addShadowCaster(mesh);mesh.receiveShadows=true;this.meshes.push(mesh);b.placed=true;b.claimedBy=null;
    const completed=this.blocks.filter(x=>x.placed).length,ratio=completed/this.blocks.length;const phase=ratio<.18?'FOUNDATION':ratio<.72?'STRUCTURE':ratio<.92?'ROOF':'DETAILS';updateWorldLabel(this.label,`${this.definition.label}\n${phase} ${Math.floor(ratio*100)}%`);
    if(completed===this.blocks.length)this.finish();
  }
  finish(){
    if(this.done)return;this.done=true;if(this.marker)this.marker.dispose();disposeWorldLabel(this.label);const b=new Building(this.kingdom,this.type,this.pos,this.meshes,this.rotation,this.definition,this.variant);buildings.push(b);
    this.kingdom.housing+=this.definition.capacity||0;if(this.type==='TownHall'){this.kingdom.townHall=b;this.kingdom.storagePos=b.pos.clone();this.kingdom.housing+=4;}if(this.type==='Keep')this.kingdom.capital=b;
    connectRoad(this.kingdom,b);addEvent(`${this.kingdom.name} completed ${this.definition.label}.`,this.kingdom.id);updateStage(this.kingdom);
  }
}
