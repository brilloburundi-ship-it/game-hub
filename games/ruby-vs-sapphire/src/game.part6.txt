function assignTask(n){
  const k=n.kingdom;
  if(n.role==='builder'||(k.projectCount()>0&&Math.random()<.16)){
    for(const p of projects){if(p.kingdom!==k||p.done)continue;const claim=p.claim(n);if(claim){n.claim=claim;n.target=k.storagePos.clone();n.task='fetchBuild';n.role='builder';return;}}
  }
  if(k.has('Farm')&&k.resources.food<k.population()*2.4&&Math.random()<.25){n.role='farmer';n.target=k.structure('Farm');n.task='workFarm';return;}
  if(k.has('Windmill')&&Math.random()<.08){n.role='miller';n.target=k.structure('Windmill');n.task='workMill';return;}
  let type;if(k.resources.food<Math.max(38,k.population()*2.0))type='food';else if(k.resources.wood<50)type='wood';else if(k.resources.stone<52)type='stone';else if(k.era>=3&&k.resources.iron<18)type='iron';else type=pick(['wood','food','stone']);
  let r=closestResource(type,n.pos(),92,n);if(!r&&type==='iron')r=closestResource('stone',n.pos(),92,n);if(r){r.reservedBy=n;n.target=r;n.task='gather';n.role=type==='wood'?'lumberjack':type==='food'?'forager':'miner';return;}
  if(!n.wanderTarget||dist2(n.pos(),n.wanderTarget)<1)n.wanderTarget=nearestLand(k.center.add(v3(rand(-12,12),0,rand(-12,12))),4);n.moveTo(n.wanderTarget,CONFIG.simTick,.3);
}

  const effects=buildings.filter(b=>b.kingdom===k&&!b.destroyed).reduce((s,b)=>s+(b.definition.effects?.tech||0),0);k.tech+=dt*(.42+k.population()*.014+effects*.18);
  const th=[0,0,85,245,520,940];while(k.era<5&&k.tech>=th[k.era+1]){k.era++;addEvent(`${k.name} entered ${['','Survival','Agriculture','Craftsmanship','Metallurgy','Kingdom'][k.era]} Era.`,k.id)}
}
function economy(k,dt){
  const farms=k.count('Farm'),mills=k.count('Windmill'),docks=k.count('Dock'),lumber=k.count('LumberCamp'),quarry=k.count('Quarry'),mine=k.count('Mine');
  k.resources.food+=(farms*.055+mills*.085+docks*.035)*dt;k.resources.wood+=lumber*.028*dt;k.resources.stone+=quarry*.020*dt;k.resources.iron+=mine*.008*dt;
  k.foodTimer+=dt;if(k.foodTimer>5){k.foodTimer=0;k.resources.food=Math.max(0,k.resources.food-k.population()*.16)}
}
